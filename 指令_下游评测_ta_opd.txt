#!/usr/bin/env bash
# 下游基准评测 (TA-OPD 论文 Table 3 口径) —— 本机 8x4090 并行版
# 说明:
#   - 训练阶段已完成 (smoke / main_sweep / diagnostics)，本步只做「下游评测 + 聚合」。
#   - 权重转换 (torch_dist -> HF) 在先前步骤已完成，main_sweep 下多数 run 已有
#     fixed_context/student_hf (或 student_hf) 可直接被 lm-eval 加载。
#   - 论文 Table 3 全部方法要求 10% budget，但本地训练仅 pure_opd(full) 有 10%，
#     其余方法最高只到 1.0% / 0.5%。故下表按「各方法可用的最高 budget」评测，
#     并在聚合表标注 budget，避免与论文同 budget 直接比较时误读。
#   - GPQA-Diamond (Idavidrein/gpqa) 为 gated 数据集，需要 HF token。
#     本机当前无 HF_TOKEN，故先评 5 个可下载基准；GPQA 待 token 就绪后单独补跑再聚合
#     (见文件末尾注释块)。
#   - 2026-07-31 更新: 改为 few-shot 评测 (MATH-500 / AIME 用 --num_fewshot 4, 带 \boxed{} 示例)。
#     原因: 学生为 Qwen3-1.7B-Base, 0-shot 时 MATH-500 exact_match≈0.4%、AIME≈0
#     (base 模型不输出 \boxed{} 答案), 属评测协议问题而非方法问题。few-shot 调用失败会自动回退 0-shot。
#     重跑前请先清旧产物: `FORCE_RERUN=1 bash 指令_下游评测_ta_opd.txt` (会删除既有 seed 目录)。
# 环境: conda env=lmeval ; 数据集缓存置于 dk/data/hf_cache
set -uo pipefail
source /opt/conda/etc/profile.d/conda.sh
conda activate lmeval

EXISTING_ROOT=/inspire/hdd/project/multi-agent/zhangweinan-24046/dk
STUDENT_HF=${EXISTING_ROOT}/modelweights/Qwen3-1.7B-Base
EVAL_ROOT=${EXISTING_ROOT}/outputs/downstream_eval
OUT=${EXISTING_ROOT}/outputs/main_sweep

export HF_ENDPOINT="${HF_ENDPOINT:-https://hf-mirror.com}"
export HF_HOME=${EXISTING_ROOT}/data/hf_cache
export HF_HUB_CACHE=${HF_HOME}/hub
export HF_DATASETS_CACHE=${HF_HOME}/datasets
export HF_ALLOW_CODE_EVAL=1   # humaneval 的 code_eval 指标需要执行生成代码
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True   # 减少多实例共享 GPU 时的显存碎片 OOM

# 5 个可下载基准 (gpqa_diamond 因 gated 暂略)
TASKS="aime24,aime25,humaneval,ifeval,hendrycks_math500"
SEEDS=(0 1 2 3 4)

# ---- 评测模型清单 (method|HF路径|budget) ----
# budget 仅用于记录/标注，不影响评测过程
MODELS=(
  "base|${STUDENT_HF}|base"
  "pure_opd|${OUT}/k16_ratio10_seed2_20260725/full_max64_seed2/student_hf|10%"
  "ta_opd|${OUT}/k16_ratio005_seed1_20260725/dlearn_high_max64_seed1/fixed_context/student_hf|0.5%"
  "entropy|${OUT}/k16_ratio010_seed2_20260725/entropy_max64_seed2/fixed_context/student_hf|1%"
  "tip|${OUT}/k16_ratio010_seed2_20260725/tip_max64_seed2/fixed_context/student_hf|1%"
  "ta_opd_entropy|${OUT}/k16_ratio010_seed2_20260725/ca_softor_max64_seed2/fixed_context/student_hf|1%"
)

mkdir -p ${EVAL_ROOT}

# ---- 预取数据集 (单进程 limit 1)，避免并行评测时并发写 HF cache 冲突 ----
echo "[prefetch] 下载 5 个基准数据集 ..."
CUDA_VISIBLE_DEVICES=0 lm-eval --model hf \
  --model_args "pretrained=${STUDENT_HF},dtype=bfloat16,trust_remote_code=True" \
  --tasks ${TASKS} --device cuda:0 --batch_size 8 --seed 0 --limit 1 \
  --confirm_run_unsafe_code --cache_requests \
  --output_path ${EVAL_ROOT}/_prefetch >/dev/null 2>&1 || true
echo "[prefetch] 完成 (忽略 limit 警告)"

# 单次 lm-eval 调用; 参数: gpu model seed tasks extra out log
run_lmeval() {
  local gpu="$1" model="$2" seed="$3" tasks="$4" extra="$5" out="$6" log="$7"
  mkdir -p "${out}"
  CUDA_VISIBLE_DEVICES="${gpu}" lm-eval --model hf \
    --model_args "pretrained=${model},dtype=bfloat16,trust_remote_code=True" \
    --tasks "${tasks}" ${extra} \
    --device cuda:0 --batch_size 8 --seed "${seed}" \
    --confirm_run_unsafe_code --cache_requests \
    --output_path "${out}" > "${log}" 2>&1
  return $?
}

run_one() {
  local name="$1" gpu="$2" model="$3" seed="$4"
  local dir="${EVAL_ROOT}/${name}/seed${seed}"
  # FORCE_RERUN=1 时强制清掉旧目录 (旧 0-shot 顶层 results 会与新的 few-shot 子目录混在一起聚合)
  if [ -n "${FORCE_RERUN:-}" ]; then rm -rf "${dir}"; fi
  [ -f "${dir}/DONE" ] && { echo "[skip] ${name} seed${seed}"; return; }
  mkdir -p "${dir}"
  # 清掉上一次协议的残留产物, 避免新旧结果混合
  rm -rf "${dir}/fs_math" "${dir}/fs_aime" "${dir}/default_other" "${dir}"/results_*.json "${dir}"/run_*.log
  echo "[$(date +%T)] START ${name} seed=${seed} gpu=${gpu}"
  local ec=0
  # (A) MATH-500: 必须 few-shot (带 \boxed{} 示例), 否则 base 模型 0-shot 几乎全错 -> 失败回退 0-shot
  run_lmeval "${gpu}" "${model}" "${seed}" "hendrycks_math500" "--num_fewshot 4 --fewshot_as_seed" "${dir}/fs_math" "${dir}/run_math.log" \
    || run_lmeval "${gpu}" "${model}" "${seed}" "hendrycks_math500" "" "${dir}/fs_math" "${dir}/run_math.log"
  ec=$((ec + $?))
  # (B) AIME24/25: 尝试 few-shot, 失败回退 0-shot
  run_lmeval "${gpu}" "${model}" "${seed}" "aime24,aime25" "--num_fewshot 4 --fewshot_as_seed" "${dir}/fs_aime" "${dir}/run_aime.log" \
    || run_lmeval "${gpu}" "${model}" "${seed}" "aime24,aime25" "" "${dir}/fs_aime" "${dir}/run_aime.log"
  ec=$((ec + $?))
  # (C) HumanEval / IFEval: 默认协议 (代码/指令遵循, 0-shot 即可)
  run_lmeval "${gpu}" "${model}" "${seed}" "humaneval,ifeval" "" "${dir}/default_other" "${dir}/run_other.log"
  ec=$((ec + $?))
  if [ ${ec} -eq 0 ]; then date '+%F_%T' > "${dir}/DONE"; echo "[OK] ${name} seed${seed}"; else echo "[FAIL] ${name} seed${seed} ec=${ec}"; fi
}

# ---- 构建任务队列 ----
jobs=()
for entry in "${MODELS[@]}"; do
  IFS='|' read -r mname mpath mbudget <<< "${entry}"
  if [ ! -d "${mpath}" ]; then echo "[WARN] 模型目录缺失: ${mpath}"; continue; fi
  for s in "${SEEDS[@]}"; do jobs+=("${mname}|${mpath}|${s}"); done
done

# ---- 并行执行 (严格 8 卡并发：每卡同一时刻只跑 1 个，避免显存叠加 OOM) ----
# 旧版用 wait -n 仅等一个就继续投新任务，导致 30 个任务在 8 卡上堆叠 (~4/卡) 触发 OOM。
# 现改为「分批 + 整批 wait」：每批最多 MAX 个 (每卡 1 个)，整批跑完再进下一批。
MAX=8
total=${#jobs[@]}
for ((i=0; i<total; i+=MAX)); do
  batch=("${jobs[@]:i:MAX}")
  echo "[round $((i/MAX+1))] 启动 ${#batch[@]} 个 (共 ${total} 个)"
  gi=0
  for j in "${batch[@]}"; do
    gpu=$((gi % MAX)); gi=$((gi+1))
    IFS='|' read -r mname mpath s <<< "${j}"
    run_one "${mname}" "${gpu}" "${mpath}" "${s}" &
  done
  wait   # 等本批 8 卡全部完成再进下一批，确保每卡同时只跑一个
  echo "[round $((i/MAX+1))] 完成"
done
echo "===== 评测完成 (5 基准: ${TASKS}) ====="

# ===== GPQA 补跑说明（待 HF_TOKEN 就绪后执行）=====
# 若已配置 HF_TOKEN：
#   export HF_TOKEN=...
#   CUDA_VISIBLE_DEVICES=0 lm-eval --model hf \
#     --model_args "pretrained=<模型>,dtype=bfloat16,trust_remote_code=True" \
#     --tasks gpqa_diamond --device cuda:0 --batch_size auto --seed <S> \
#     --output_path ${EVAL_ROOT}/<name>/seed<S>
# 然后更新聚合脚本把 gpqa_diamond 也纳入 (BENCH_TASK 增加 gpqa_diamond)。

# 若仍接近 0, 说明 few-shot 未生效或模型未输出 \boxed{}, 需换用 --apply_chat_template / 自定义 prompt 模板。
